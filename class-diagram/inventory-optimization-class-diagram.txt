package "Entity Layer" <<entity>> {

    class InventoryParameter <<entity>> {
        -id: Long
        -productId: String
        -warehouseConfig: WarehouseConfig
        -supplierProductId: UUID
        -planStartDate: LocalDate
        -planEndDate: LocalDate
        -scheduleStartDate: LocalDate
        -demandQ: BigDecimal
        -snapshotSupplyRateK: BigDecimal
        -snapshotFixedOrderCostA: BigDecimal
        -snapshotUnitPriceC: BigDecimal
        -snapshotLeadTimeL: BigDecimal
        -storageCostCoefficientI: BigDecimal
    }

    class ConsumptionHistory <<entity>> {
        -id: Long
        -productId: String
        -periodStartDate: LocalDate
        -periodEndDate: LocalDate
        -actualConsumption: BigDecimal
        -plannedConsumption: BigDecimal
        -actualLeadTimeDays: BigDecimal
        -actualSupplyRate: BigDecimal
        -notes: String
        -createdAt: LocalDateTime
    }

    class InventoryResult <<entity>> {
        -id: Long
        -inventoryParameter: InventoryParameter
        -optimalOrderQtyS: BigDecimal
        -optimalOrderCountN: BigDecimal
        -optimalCycleTimeTau: BigDecimal
        -maxInventoryLevel: BigDecimal
        -avgInventoryLevel: BigDecimal
        -reorderPointB: BigDecimal
        -minInventoryCost: BigDecimal
    }

    class WarehouseConfig <<entity>> {
        -id: Long
        -interestRate: BigDecimal
        -warehouseMonthlyCost: BigDecimal
        -warehouseMaxCapacity: BigDecimal
        -spoilageRate: BigDecimal
        -insuranceRate: BigDecimal
        -storageCostCoefficient: BigDecimal
        -updatedAt: LocalDateTime
    }

    class OrderSchedule <<entity>> {
        -id: Long
        -inventoryResult: InventoryResult
        -productId: String
        -orderSequence: Integer
        -orderDate: LocalDate
        -expectedDeliveryDate: LocalDate
        -orderQuantity: BigDecimal
        -estimatedCost: BigDecimal
        -isReorderWarning: Boolean
    }

    ' Layout: Xếp dọc các class có quan hệ chứa nhau để thu hẹp chiều ngang
    InventoryParameter -[hidden]right- ConsumptionHistory
    InventoryResult    -[hidden]right- WarehouseConfig
    
    InventoryParameter -[hidden]down- InventoryResult
    ConsumptionHistory -[hidden]down- WarehouseConfig
    InventoryResult    -[hidden]down- OrderSchedule

    ' Composition relationships (Đi thẳng trục dọc)
    InventoryParameter "1" *-down- "*" InventoryResult
    InventoryResult    "1" *-down- "*" OrderSchedule
}
