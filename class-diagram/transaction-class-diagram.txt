' ==================== LAYER 1: CONTROLLER ====================

        +getAllTransactions()

        +getTransactionById()

' ==================== LAYER 6: ENTITY ====================
' 3 entity xếp ngang — đồng nhất với repository
package "Entity Layer" <<entity>> {

    class InOutTransaction <<entity>> {
        -id: String (UUID)
        -warehouseId: String
        -workType: WorkType
        -status: TransactionStatus
        -createdAt: LocalDateTime
        -confirmedAt: LocalDateTime
        -updatedAt: LocalDateTime
        -inOutDetails: Set<InOutDetail>
        -transactionFile: TransactionFile
    }

    class InOutDetail <<entity>> {
        -id: String (UUID)
        -transaction: InOutTransaction
        -productId: String
        -quantity: BigDecimal
        -createdAt: LocalDateTime
        -updatedAt: LocalDateTime
    }

    class TransactionFile <<entity>> {
        -id: Long (AUTO)
        -fileName: String
        -objectName: String
        -contentType: String
        -fileSize: Long
        -uploadedAt: LocalDateTime
        -transaction: InOutTransaction
    }

    InOutTransaction -[hidden]right- InOutDetail
    InOutDetail      -[hidden]right- TransactionFile

    InOutTransaction "1" *-down- "*"    InOutDetail
    InOutTransaction "1" *-down- "0..1" TransactionFile
}

' ==================== LAYER 7: DTO ====================
' Chia làm 2 hàng, mỗi hàng tối đa 2 class → bề ngang hẹp hơn nhiều
package "DTO Layer" <<dto>> {

    ' Hàng 1: Request types (2 class)
    class InOutTransactionRequest <<dto>> {
        +warehouseId: String
        +workType: WorkType
        +inOutDetails: List<InOutDetailRequest>
    }

    class InOutTransactionUpdateRequest <<dto>> {
        +warehouseId: String
        +workType: WorkType
        +inOutDetails: List<InOutDetailRequest>
    }

    ' Hàng 2: Detail request + response (2 class)
    class InOutDetailRequest <<dto>> {
        +productId: String
        +quantity: BigDecimal
    }

    class InOutDetailResponse <<dto>> {
        +id: String
        +productId: String
        +productName: String
        +quantity: BigDecimal
    }

    ' Hàng 3: Transaction response + Report (2 class)
    class InOutTransactionResponse <<dto>> {
        +id: String
        +warehouseId: String
        +workType: WorkType
        +status: TransactionStatus
        +createdAt: LocalDateTime
        +confirmedAt: LocalDateTime
        +updatedAt: LocalDateTime
        +inOutDetails: List<InOutDetailResponse>
    }

    class TransactionReport <<dto>> {
        +id: String
        +warehouseId: String
        +workType: WorkType
        +status: TransactionStatus
        +reportDetails: List<ReportDetail>
    }

    ' Hàng 4: các DTO phụ (2 class)

    ' Layout: 2 cột, 4 hàng
    InOutTransactionRequest    -[hidden]right- InOutTransactionUpdateRequest
    InOutDetailRequest         -[hidden]right- InOutDetailResponse
    InOutTransactionResponse   -[hidden]right- TransactionReport

    InOutTransactionRequest    -[hidden]down-  InOutDetailRequest
    InOutTransactionUpdateRequest -[hidden]down- InOutDetailResponse
    InOutDetailRequest         -[hidden]down-  InOutTransactionResponse
    InOutDetailResponse        -[hidden]down-  TransactionReport
}
