@startuml
!theme plain

actor "Quản lý kho" as Manager
boundary "Browser\n(Trang Sản phẩm)" as Browser
boundary "ProductForm\nComponent" as Form
control "ProductController" as Controller
database "Database\n(Product)" as Database

Manager -> Browser: Truy cập trang\n"Sản phẩm"
activate Browser

Browser -> Manager: Hiển thị danh sách\nsản phẩm

Manager -> Browser: Chọn sản phẩm\nmuốn sửa

Manager -> Browser: Nhấn nút "Sửa"

Browser -> Form: Mở form sửa\nsản phẩm
activate Form

Form -> Manager: Hiển thị form với\nthông tin sản phẩm\nđã được điền sẵn

Manager -> Form: Chỉnh sửa thông tin\nsản phẩm

Manager -> Form: (Tùy chọn) Thay đổi\nảnh sản phẩm

alt Quản lý kho hủy thao tác
  Manager -> Form: Nhấn "Hủy"
  Form -> Form: Đóng form
  deactivate Form
  
  note over Browser
    Không lưu thay đổi
    Quay lại trang "Sản phẩm"
  end note
else Quản lý kho xác nhận cập nhật
  Manager -> Form: Nhấn "Xác nhận"
  activate Form
  
  alt Dữ liệu không hợp lệ
    Form -> Form: Kiểm tra validation\n(tên/đơn vị trống,...)
    Form -> Manager: Hiển thị thông báo lỗi
    
    Manager -> Form: Chỉnh sửa thông tin\nvà xác nhận lại
  else Dữ liệu hợp lệ
    Form -> Controller: PUT /api/products/{id}
    activate Controller
    
    Controller -> Database: Cập nhật thông tin\nsản phẩm
    activate Database
    Database -> Database: Lưu\nthay đổi
    Database --> Controller: ProductResponse
    deactivate Database
    
    Controller --> Form: HTTP 200 OK\n(ProductResponse)
    deactivate Controller
    
    opt Nếu ảnh được thay đổi
      Form -> Controller: POST /api/products/{id}/\nupload-image
      activate Controller
      
      Controller -> Database: Cập nhật\nảnh sản phẩm
      activate Database
      Database -> Database: Lưu file\nảnh mới
      Database --> Controller: HTTP 200 OK
      deactivate Database
      
      Controller --> Form: HTTP 200 OK
      deactivate Controller
    end
    
    Form -> Form: Đóng form
    
    Form -> Browser: Refresh danh sách
    activate Browser
    
    Browser -> Browser: Cập nhật danh sách
    Browser -> Manager: Hiển thị danh sách\nsản phẩm
    deactivate Browser
  end
end

deactivate Form

@enduml
