@startuml
!theme plain

actor "Quản lý kho" as Manager
boundary "Browser\n(Trang Kho hàng)" as Browser
boundary "WarehouseForm\nComponent" as Form
control "WarehouseController" as Controller
database "Database\n(Warehouse)" as Database

Manager -> Browser: Truy cập trang\n"Kho hàng"
activate Browser

Browser -> Manager: Hiển thị danh sách\nkho hàng

Manager -> Browser: Chọn kho hàng\ncần sửa

Manager -> Browser: Nhấn nút "Sửa"

Browser -> Form: Mở form sửa\nkho hàng
activate Form

Form -> Controller: GET /api/warehouses/\nfull-info/{id}
activate Controller

Controller -> Database: Truy vấn thông tin\nkho hàng
activate Database
Database --> Controller: FullWarehouseResponse
deactivate Database

Controller --> Form: HTTP 200 OK\n(Thông tin kho hàng)
deactivate Controller

Form -> Manager: Hiển thị form với\nthông tin kho hàng\nđã được điền sẵn

Manager -> Form: Chỉnh sửa thông tin\nkho hàng

alt Quản lý kho hủy thao tác
  Manager -> Form: Nhấn "Hủy"
  Form -> Form: Đóng form
  deactivate Form
  
  note over Browser
    Quay lại trang "Kho hàng"
    Không lưu thay đổi
  end note
else Quản lý kho xác nhận cập nhật
  Manager -> Form: Nhấn "Xác nhận"
  activate Form
  
  alt Dữ liệu không hợp lệ
    Form -> Form: Kiểm tra validation
    Form -> Manager: Hiển thị thông báo lỗi\n(tên trống, định dạng sai,\nsố âm,...)
    
    Manager -> Form: Chỉnh sửa thông tin\nvà xác nhận lại
  else Dữ liệu hợp lệ
    Form -> Controller: PUT /api/warehouses
    activate Controller
    
    Controller -> Database: Cập nhật thông tin\nkho hàng
    activate Database
    Database -> Database: Lưu thay đổi\nthông tin kho hàng
    Database --> Controller: WarehouseResponse
    deactivate Database
    
    Controller --> Form: HTTP 200 OK
    deactivate Controller
    
    Form -> Form: Đóng form
    
    Form -> Browser: Refresh danh sách
    activate Browser
    
    Browser -> Browser: Cập nhật danh sách
    Browser -> Manager: Hiển thị danh sách\nkho hàng với thông tin\nmới của kho hàng\nvừa được sửa
    deactivate Browser
  end
end

deactivate Form

@enduml
