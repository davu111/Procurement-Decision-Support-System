@startuml
!theme plain

actor "Quản lý kho" as Manager
boundary "Browser\n(Trang Nhà cung cấp)" as Browser
boundary "SupplierForm\nComponent" as Form
control "SupplierController" as Controller
database "Database\n(Supplier)" as Database

Manager -> Browser: Truy cập trang\n"Nhà cung cấp"
activate Browser

Browser -> Manager: Hiển thị danh sách\nnhà cung cấp

Manager -> Browser: Nhấn nút "Thêm"

Browser -> Form: Mở form tạo\nnhà cung cấp mới
activate Form

Form -> Manager: Hiển thị form nhập\nthông tin nhà cung cấp

Manager -> Form: Nhập đầy đủ\nthông tin nhà cung cấp

alt Quản lý kho hủy thao tác
  Manager -> Form: Nhấn "Hủy"
  Form -> Form: Đóng form
  deactivate Form
  
  note over Browser
    Không lưu dữ liệu
    Quay lại trang "Nhà cung cấp"
  end note
else Quản lý kho xác nhận tạo
  Manager -> Form: Nhấn "Xác nhận"
  activate Form
  
  alt Dữ liệu không hợp lệ
    Form -> Form: Kiểm tra validation\n(mã/tên trống, email sai\nđịnh dạng,...)
    Form -> Manager: Hiển thị thông báo lỗi
    
    Manager -> Form: Chỉnh sửa thông tin\nvà xác nhận lại
  else Dữ liệu hợp lệ
    Form -> Controller: POST /api/suppliers
    activate Controller
    
    Controller -> Database: Tạo nhà\ncung cấp mới
    activate Database
    Database -> Database: Lưu\nthông tin
    Database --> Controller: Supplier\nResponse
    deactivate Database
    
    Controller --> Form: HTTP 201 Created\n(SupplierResponse)
    deactivate Controller
    
    Form -> Form: Đóng form
    
    Form -> Browser: Refresh danh sách
    activate Browser
    
    Browser -> Browser: Cập nhật danh sách
    Browser -> Manager: Hiển thị danh sách\nnhà cung cấp mới
    deactivate Browser
  end
end

deactivate Form

@enduml
