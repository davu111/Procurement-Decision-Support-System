@startuml
!theme plain

actor "Quản lý kho" as QL
boundary "Browser\n(Trang Kiểm kê kho)" as Browser
boundary "StockCountForm\nComponent" as Form
boundary "StockCountDetail\nDialog" as Dialog
control "StockCountController" as Controller
database "Database\n(StockCount)" as Database

QL -> Browser: Truy cập trang\n"Kiểm kê kho"
activate Browser

Browser -> QL: Hiển thị trang danh sách\nphiếu kiểm kê

QL -> Browser: Chọn sản phẩm,\nngày kiểm kê,\nnguời thực hiện

Browser -> Form: Cập nhật thông tin\nlọc
activate Form

Form -> QL: Hiển thị danh sách\nphiếu theo lọc
deactivate Form

QL -> Browser: Nhấn nút\n"Tạo phiếu tạm"

Browser -> Form: Mở form tạo\nphiếu kiểm kê mới
activate Form

Form -> Controller: Gọi API tạo phiếu\nPOST /api/stock-count\nvới thông tin chọn lọc
activate Controller

Controller -> Database: Tạo phiếu kiểm kê\ntrạng thái DRAFT\nlưu sản phẩm, ngày, người
activate Database
Database --> Controller: StockCountResponse
deactivate Database

Controller --> Form: HTTP 201 Created
deactivate Controller

Form -> Dialog: Mở dialog\nchi tiết phiếu kiểm kê
activate Dialog
deactivate Form

Dialog -> QL: Hiển thị thông tin phiếu\ntrạng thái "Chờ xác nhận"

QL -> Dialog: Nhấn nút\n"Xác nhận"

Dialog -> Dialog: Chuyển trạng thái\nthành "Đang kiểm kê"

Dialog -> QL: Hiển thị form\nnhập số lượng kiểm kê

QL -> Dialog: Nhập số lượng\nthực tế kiểm kê được

alt Quản lý kho hủy thao tác
  QL -> Dialog: Nhấn nút "Hủy"
  
  Dialog -> Controller: Gọi API xóa phiếu\nDELETE /api/stock-count/{id}
  activate Controller
  
  Controller -> Database: Xóa phiếu kiểm kê\ntrạng thái DRAFT
  activate Database
  Database --> Controller: HTTP 204
  deactivate Database
  
  Controller --> Dialog: Xóa thành công
  deactivate Controller
  
  Dialog -> Dialog: Đóng dialog
  deactivate Dialog
  
  Browser -> QL: Quay lại trang\n"Kiểm kê kho"
  
  note over Browser
    Luồng thay thế 6a:
    Hệ thống không lưu phiếu kiểm kê.
    Hệ thống quay lại trang "Kiểm kê kho".
  end note
  
else Quản lý kho xác nhận lưu
  QL -> Dialog: Nhấn nút "Lưu"
  
  Dialog -> Form: Xác nhận lưu phiếu
  activate Form
  
  alt Dữ liệu không hợp lệ
    Form -> Form: Kiểm tra validation\n(số lượng âm, thiếu dữ liệu)
    Form -> QL: Hiển thị thông báo lỗi
    
    QL -> Dialog: Chỉnh sửa lại\nthông tin
  else Dữ liệu hợp lệ
    Form -> Controller: Gọi API lưu phiếu\nPUT /api/stock-count/{id}\nvới số lượng kiểm kê
    activate Controller
    
    Controller -> Database: Cập nhật phiếu kiểm kê\ntrạng thái COMPLETED\nlưu số lượng thực tế
    activate Database
    Database --> Controller: StockCountResponse
    deactivate Database
    
    Controller --> Form: HTTP 200 OK
    deactivate Controller
    
    Form -> Dialog: Phiếu được lưu thành công
    deactivate Form
    
    Dialog -> Dialog: Đóng dialog
    deactivate Dialog
    
    Browser -> Browser: Refresh danh sách\nphiếu kiểm kê
    
    Browser -> QL: Hiển thị phiếu kiểm kê\nđã lưu trong danh sách
    
    note over QL
      Luồng chính 7:
      Phiếu kiểm kê được lưu vào hệ thống.
      Kết quả kiểm kê được ghi nhận để đối chiếu
      và xử lý chênh lệch tồn kho.
      Kết thúc Use Case.
    end note
  end
end

@enduml
