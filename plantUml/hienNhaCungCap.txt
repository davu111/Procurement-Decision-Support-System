@startuml
!theme plain

actor "Quản lý kho" as Manager
boundary "Browser\n(Trang Nhà cung cấp)" as Browser
boundary "ConfirmDialog\nComponent" as Dialog
control "SupplierController" as Controller
database "Database\n(Supplier)" as Database

Manager -> Browser: Truy cập trang\n"Nhà cung cấp"
activate Browser

Browser -> Manager: Hiển thị danh sách\nnhà cung cấp

Manager -> Browser: Nhấn nút "Hiện"\ntại nhà cung cấp\nđang bị ẩn

Browser -> Dialog: Mở hộp thoại\nxác nhận
activate Dialog

Dialog -> Manager: Hiển thị thông báo\n"Bạn có chắc chắn\nmuốn hiện nhà cung cấp này?"

alt Quản lý kho hủy thao tác
  Manager -> Dialog: Nhấn "Hủy"
  Dialog -> Dialog: Đóng hộp thoại
  deactivate Dialog
  
  note over Browser
    Nhà cung cấp không được hiển thị lại
    Quay lại danh sách
  end note
else Quản lý kho xác nhận hiện
  Manager -> Dialog: Nhấn "Xác nhận"
  activate Dialog
  
  Dialog -> Controller: PATCH /api/suppliers/\nactive/{id}
  activate Controller
  
  Controller -> Database: Cập nhật\ntrạng thái
  activate Database
  Database -> Database: Thay đổi\nisActive = true
  Database --> Controller: HTTP 200 OK
  deactivate Database
  
  Controller --> Dialog: HTTP 200 OK
  deactivate Controller
  
  Dialog -> Dialog: Đóng hộp thoại
  
  Dialog -> Browser: Refresh danh sách
  activate Browser
  
  Browser -> Browser: Cập nhật danh sách
  Browser -> Manager: Hiển thị danh sách\nnhà cung cấp
  deactivate Browser
end

deactivate Dialog

@enduml
