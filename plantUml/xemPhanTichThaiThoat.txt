@startuml
!theme plain

actor "Quản lý kho" as QL
boundary "Browser\n(Trang Phân tích thất thoát)" as Browser
boundary "WasteLossAnalysisPage\nComponent" as Page
participant "AnalyticsController" as Controller
database "Database\n(StockCount, Transaction)" as Database

QL -> Browser: Truy cập trang\n"Phân tích thất thoát"
activate Browser

Browser -> Page: Render trang Phân tích\nGET /api/analytics/waste-loss\n?from=...&to=...
activate Page

Page -> Controller: Gọi API lấy dữ liệu\nphân tích (mặc định khoảng\nthời gian: 30 ngày gần nhất)
activate Controller

Controller -> Database: Truy vấn dữ liệu kiểm kê,\nnhập kho, xuất kho,\ntiêu thụ thực tế\ntrong khoảng thời gian
activate Database
Database --> Controller: Trả về dữ liệu phân tích
deactivate Database

Controller -> Controller: Tính toán tỷ lệ thất thoát\nthất thoát = (Tồn kho lý thuyết\n- Tồn kho thực tế) / Tồn kho lý thuyết * 100%

alt Không có dữ liệu trong khoảng thời gian
  Controller --> Page: Dữ liệu rỗng
  
  Page -> Page: Hiển thị thông báo\n"Không có dữ liệu để phân tích"
  
  Page --> Browser: Thông báo không có dữ liệu
  deactivate Page
  
  Browser --> QL: Hiển thị thông báo\n"Chưa có dữ liệu kiểm kê\ntrong khoảng thời gian này"
  
  note over QL
    Luồng ngoại lệ 3a:
    Không có dữ liệu trong khoảng thời gian được chọn.
    Kết thúc Use Case.
  end note
  
else Có dữ liệu để phân tích
  Controller --> Page: ApiResponse<WasteLossAnalysisDto>
  deactivate Controller
  
  Page -> Page: Hiển thị biểu đồ phân tích:\n- Tỷ lệ thất thoát theo sản phẩm\n- Xu hướng thất thoát theo thời gian\n- So sánh lý thuyết vs thực tế
  
  Page -> Page: Hiển thị bảng thống kê:\n- Sản phẩm, Tồn kho lý thuyết,\n- Tồn kho thực tế, Chênh lệch,\n- Tỷ lệ thất thoát (%)
  
  Page --> Browser: Hiển thị trang Phân tích\nvới biểu đồ và bảng thống kê
  deactivate Page
  
  Browser --> QL: Hiển thị các số liệu thống kê\nvà biểu đồ phân tích thất thoát
end

QL -> Browser: Chọn sản phẩm cụ thể\nvà khoảng thời gian\nmong muốn để phân tích
activate Browser

Browser -> Page: Trigger filter change event\nwith productId, startDate, endDate
activate Page

Page -> Controller: Gọi API lấy dữ liệu\nphân tích có lọc\nGET /api/analytics/waste-loss\n?productId=...&from=...&to=...
activate Controller

Controller -> Database: Truy vấn dữ liệu kiểm kê,\nnhập kho, xuất kho\nvới bộ lọc sản phẩm\nvà thời gian
activate Database
Database --> Controller: Trả về dữ liệu lọc
deactivate Database

Controller -> Controller: Tính toán tỷ lệ thất thoát\nvới dữ liệu lọc

Controller --> Page: ApiResponse<WasteLossAnalysisDto>
deactivate Controller

Page -> Page: Cập nhật biểu đồ\nvà bảng thống kê\nvới dữ liệu lọc

Page --> Browser: Cập nhật hiển thị
deactivate Page

Browser --> QL: Hiển thị kết quả phân tích\nthất thoát theo sản phẩm\nvà khoảng thời gian đã chọn
deactivate Browser

note over QL
  Luồng chính 4-5:
  Quản lý kho xem kết quả phân tích.
  Dữ liệu hiển thị bao gồm:
  - Tỷ lệ thất thoát (%)
  - Xu hướng theo thời gian
  - So sánh dự kiến vs thực tế
  - Các sản phẩm có tỷ lệ thất thoát cao
  Kết thúc Use Case.
end note

@enduml
