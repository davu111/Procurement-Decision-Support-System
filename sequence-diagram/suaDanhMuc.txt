@startuml
!theme plain

actor "Quản lý kho" as Manager
boundary "Browser\n(Trang Danh mục)" as Browser
boundary "CategoryForm\nComponent" as Form
control "ProductCategoryController" as Controller
database "Database\n(ProductCategory)" as Database

Manager -> Browser: Truy cập trang\n"Danh mục"
activate Browser

Browser -> Manager: Hiển thị danh sách\ndanh mục

Manager -> Browser: Chọn danh mục\nmuốn sửa

Manager -> Browser: Nhấn nút "Sửa"

Browser -> Form: Mở form sửa\ndanh mục
activate Form

Form -> Manager: Hiển thị form với\nthông tin danh mục

Manager -> Form: Chỉnh sửa thông tin\ndanh mục

alt Quản lý kho hủy thao tác
  Manager -> Form: Nhấn "Hủy"
  Form -> Form: Đóng form
  deactivate Form
  
  note over Browser
    Không lưu thay đổi
    Quay lại trang "Danh mục"
  end note
else Quản lý kho xác nhận cập nhật
  Manager -> Form: Nhấn "Xác nhận"
  activate Form
  
  alt Dữ liệu không hợp lệ
    Form -> Form: Kiểm tra validation\n(tên trống, tên quá dài,\nmô tả quá dài,...)
    Form -> Manager: Hiển thị thông báo lỗi
    
    Manager -> Form: Chỉnh sửa thông tin\nvà xác nhận lại
  else Dữ liệu hợp lệ
    Form -> Controller: PUT /api/\nproduct-categories/{id}
    activate Controller
    
    Controller -> Database: Cập nhật\nthông tin
    activate Database
    Database -> Database: Lưu\nthay đổi
    Database --> Controller: ProductCategory\nResponse
    deactivate Database
    
    Controller --> Form: HTTP 200 OK
    deactivate Controller
    
    Form -> Form: Đóng form
    
    Form -> Browser: Refresh\ndanh sách
    activate Browser
    
    Browser -> Browser: Cập nhật\ndanh sách
    Browser -> Manager: Hiển thị danh sách\ndanh mục
    deactivate Browser
  end
end

deactivate Form

@enduml
