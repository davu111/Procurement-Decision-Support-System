@startuml
!theme plain

actor "Quản lý kho" as Manager
boundary "Browser\n(Trang Kho hàng)" as Browser
boundary "TransferDialog\nComponent" as Dialog
control "WarehouseController\n& InventoryController" as Controller
database "Database\n(Warehouse, Inventory)" as Database

Manager -> Browser: Truy cập trang\n"Kho hàng"
activate Browser

Browser -> Manager: Hiển thị danh sách\nkho hàng

Manager -> Browser: Chọn kho hàng\nmuốn nhập hàng

Manager -> Browser: Nhấn nút\n"Nhập hàng"

Browser -> Dialog: Mở dialog nhập hàng
activate Dialog

Dialog -> Manager: Hiển thị form nhập\nthông tin sản phẩm

Manager -> Dialog: Nhập thông tin\nsản phẩm (sản phẩm,\nsố lượng)

Manager -> Dialog: Có thể thêm\nhoặc bỏ dòng

alt Quản lý kho hủy thao tác
  Manager -> Dialog: Nhấn "Hủy"
  Dialog -> Dialog: Đóng dialog
  deactivate Dialog
  
  note over Browser
    Không lưu dữ liệu
    Quay lại trang "Kho hàng"
  end note
else Quản lý kho xác nhận nhập hàng
  Manager -> Dialog: Nhấn "Xác nhận"
  activate Dialog
  
  alt Dữ liệu không hợp lệ
    Dialog -> Dialog: Kiểm tra validation
    Dialog -> Manager: Hiển thị thông báo lỗi\n(sản phẩm trống,\nsố lượng <= 0,...)
    
    Manager -> Dialog: Chỉnh sửa thông tin\nvà xác nhận lại
  else Dữ liệu hợp lệ
    Dialog -> Controller: POST /api/inventory-\ntransfer/transfer
    activate Controller
    
    Controller -> Database: Cập nhật số lượng\nhàng hóa trong kho
    activate Database
    Database -> Database: Tăng tồn kho\ntheo sản phẩm
    Database --> Controller: Transfer successful
    deactivate Database
    
    Dialog -> Controller: POST /api/\ntransactions
    activate Controller
    
    Controller -> Database: Tạo ghi chép\ngiao dịch nhập hàng
    activate Database
    Database -> Database: Lưu chi tiết\ngiao dịch nhập
    Database --> Controller: TransactionResponse
    deactivate Database
    
    Controller --> Dialog: HTTP 201 Created
    deactivate Controller
    
    Dialog -> Controller: POST /api/\ntransactions/{id}/\ngenerate-report
    activate Controller
    
    Controller -> Database: Tạo phiếu nhập hàng
    activate Database
    Database -> Database: Sinh file report
    Database --> Controller: FileId
    deactivate Database
    
    Controller --> Dialog: HTTP 200 OK\n(FileId)
    deactivate Controller
    
    Dialog -> Controller: GET /api/files/\n{fileId}/view-url
    activate Controller
    Controller --> Dialog: View URL
    deactivate Controller
    
    Dialog -> Controller: GET /api/files/\n{fileId}/download-url
    activate Controller
    Controller --> Dialog: Download URL
    deactivate Controller
    
    Dialog -> Manager: Hiển thị phiếu nhập hàng\n(file preview + nút tải)
    
    Manager -> Dialog: Xem hoặc\ntải về phiếu
    
    Dialog -> Dialog: Đóng dialog
    deactivate Dialog
    
    Dialog -> Browser: Refresh danh sách kho
    activate Browser
    
    Browser -> Browser: Cập nhật số lượng tồn
    Browser -> Manager: Hiển thị danh sách kho\ncập nhật
    deactivate Browser
  end
end

deactivate Dialog

@enduml
