@startuml
!theme plain

actor "Quản trị viên" as Admin
boundary "Browser\n(Trang Người dùng)" as Browser
boundary "EmployeeForm\nComponent" as Form
boundary "CredentialsDialog\nComponent" as CredDialog
control "EmployeeController" as Controller
database "Database\n(Employee)" as Database

Admin -> Browser: Truy cập trang\n"Người dùng"
activate Browser

Browser -> Admin: Hiển thị danh sách\nnhân viên

Admin -> Browser: Nhấn nút\n"Thêm người dùng"

Browser -> Form: Mở form tạo\ntài khoản mới
activate Form

Form -> Admin: Hiển thị form nhập\nthông tin nhân viên

Admin -> Form: Nhập đầy đủ\nthông tin tài khoản

alt Quản trị viên hủy thao tác
  Admin -> Form: Nhấn "Hủy"
  Form -> Form: Đóng form
  deactivate Form
  
  note over Browser
    Không lưu dữ liệu
    Quay lại trang "Người dùng"
  end note
else Quản trị viên xác nhận tạo
  Admin -> Form: Nhấn "Xác nhận"
  activate Form
  
  alt Dữ liệu không hợp lệ
    Form -> Form: Kiểm tra validation
    Form -> Admin: Hiển thị thông báo lỗi
    
    Admin -> Form: Chỉnh sửa lại thông tin\nvà xác nhận lại
  else Dữ liệu hợp lệ
    Form -> Controller: POST /api/employees
    activate Controller
    
    Controller -> Database: Tạo tài khoản mới
    activate Database
    Database -> Database: Lưu thông tin nhân viên\nTạo mật khẩu tạm thời
    Database --> Controller: EmployeeResponse
    deactivate Database
    
    Controller --> Form: HTTP 201 Created
    deactivate Controller
    
    Form -> Form: Đóng form
    
    Form -> CredDialog: Mở dialog\nhiển thị credentials
    activate CredDialog
    
    CredDialog -> Admin: Hiển thị thông tin đăng nhập
    
    Admin -> CredDialog: Xem/sao chép\nthông tin đăng nhập
    
    Admin -> CredDialog: Nhấn "Xong"
    CredDialog -> CredDialog: Đóng dialog
    deactivate CredDialog
    
    Form -> Browser: Refresh danh sách
    activate Browser
    
    Browser -> Browser: Cập nhật danh sách
    Browser -> Admin: Hiển thị danh sách\nnhân viên mới
    deactivate Browser
  end
end

deactivate Form

@enduml
